/**
   @file main.ino
   @brief Проста програма, що блимає світлодіодом на ESP32-S3.

   Цей код ініціалізує послідовний зв'язок і налаштовує пін для
   керування світлодіодом. У головному циклі програми світлодіод
   по черзі вмикається і вимикається з різними затримками.
   Це демонструє базове використання функцій `digitalWrite` та `delay`.
   @author Mykyta Siryk
   @date 2025-09-14
*/

// --- Визначення констант ---

/**
   @def ledPin
   @brief Пін, до якого під'єднано світлодіод.

   Використовує пін D5 на платі ESP32-S3.
*/
const int ledPin = 5;

// --- Функції ---

/**
   @brief Налаштування програми.

   Ця функція виконується один раз при запуску.
   Вона ініціалізує послідовний зв'язок для виведення повідомлень
   та налаштовує `ledPin` як вихідний.
*/
void setup() {
  Serial.begin(115200);
  pinMode(ledPin, OUTPUT);
  Serial.println("Hello, ESP32-S3!");
}

/**
   @brief Основний цикл програми.

   Ця функція виконується безперервно.
   Вона вмикає світлодіод на 1.5 секунди, потім вимикає його на 0.5 секунди,
   виводячи відповідні повідомлення в послідовний порт.
*/
void loop() {
  digitalWrite(ledPin, HIGH);
  Serial.println("LED is ON");
  delay(1500);

  digitalWrite(ledPin, LOW);
  Serial.println("LED is OFF");
  delay(500);
}
