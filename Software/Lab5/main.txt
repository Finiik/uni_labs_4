/**
   @file main.ino
   @brief Програма для керування світлодіодом через UART0.

   Ця програма взаємодіє з користувачем через послідовний монітор (UART0) для керування
   блиманням світлодіода. Користувач вводить число, і світлодіод, підключений до GPIO 5,
   блимає задану кількість разів. Кнопка, підключена до GPIO 4, служить для додаткової
   індикації (наприклад, виведення повідомлення про натискання в послідовний монітор).
   @author Mykyta Siryk
   @date 2025-09-14
*/

// --- Визначення констант ---

/**
   @def BUTTON_PIN
   @brief Пін GPIO, до якого підключена кнопка.
   @def LED_PIN
   @brief Пін GPIO, до якого підключений світлодіод.
*/
const int buttonPin = 4;
const int ledPin = 5;

// --- Визначення глобальних змінних ---

/**
   @var receivedMessage
   @brief Рядок для зберігання повного повідомлення, отриманого через послідовний порт.
*/
String receivedMessage = "";

/**
   @brief Функція для блимання світлодіода.

   Ця функція приймає ціле число і змушує світлодіод блимати цю кількість разів.
   Кожне блимання складається з увімкнення на 0.5 секунди та вимкнення на 0.5 секунди.
   @param numBlinks Кількість блимань.
*/
void blink(int numBlinks) {
  for (int i = 0; i < numBlinks; i++) {
    digitalWrite(ledPin, HIGH);
    delay(500);
    digitalWrite(ledPin, LOW);
    delay(500);
  }
}

/**
   @brief Функція налаштування (setup).

   Ініціалізує послідовний зв'язок на швидкості 115200 бод.
   Налаштовує піни кнопки та світлодіода:
   - `buttonPin` як вхід з внутрішнім підтягувальним резистором (`INPUT_PULLUP`).
   - `ledPin` як вихід (`OUTPUT`).
   Виводить початкове повідомлення в послідовний монітор.
*/
void setup() {
  // Старт послідовного монітора на швидкості 115200
  Serial.begin(115200);

  // Налаштування пінів
  pinMode(buttonPin, INPUT_PULLUP);
  pinMode(ledPin, OUTPUT);

  // Виведення початкового повідомлення
  Serial.println("ESP32 is ready. Please enter a number of blinks:");
}

/**
   @brief Основний цикл програми (loop).

   Безперервно перевіряє наявність даних у послідовному буфері.
   Якщо дані доступні, зчитує їх до символу нового рядка, перетворює в число,
   і викликає функцію `blink` для блимання світлодіода.
   Також відстежує стан кнопки і виводить повідомлення при її натисканні.
*/
void loop() {
  // Перевірка, чи доступні дані в послідовному буфері
  if (Serial.available() > 0) {
    // Зчитування повного рядка до символу нового рядка
    receivedMessage = Serial.readStringUntil('\n');

    // Видалення зайвих пробілів
    receivedMessage.trim();

    // Перетворення рядка на ціле число
    int numBlinks = receivedMessage.toInt();

    // Перевірка, чи число дійсне і більше 0
    if (numBlinks > 0) {
      Serial.print("Received: ");
      Serial.println(numBlinks);
      blink(numBlinks);
      Serial.println("Blinking finished. Enter a new number.");
    } else {
      Serial.println("Invalid number. Please enter a positive integer.");
    }
  }

  // Перевірка стану кнопки
  if (digitalRead(buttonPin) == LOW) {
    Serial.println("Button pressed!");
    delay(50); // Затримка для уникнення "дребезгу" контактів
  }
}
